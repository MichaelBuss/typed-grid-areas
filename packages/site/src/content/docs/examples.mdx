Two designs, the ones the library's browser tests check in every
combination of their optional areas, in every adapter. The cards on this
page are rendered with the React adapter when the site is built, as static
HTML: nothing hydrates, and no JavaScript of the library's runs in your
browser. They lay out with JavaScript turned off.

Each design is shown at {NARROW}px and {WIDE}px. Every card is a grid
measuring its own width, so the same definition gives each of them its own
arrangement.

## PreferenceCard

An optional input panel beside a content panel with its own padding. The
content panel is a [nested grid](/typed-grid-areas/concepts/nested-grids/):
an optional icon, the information, an optional tag and optional children.
With both an icon and a tag, the two share the top row above the
information on narrow cards, and from `@sm` all three share one row. With
only one of them, it sits beside the information at every width. Children
always take a full-width last row.

<PreferenceCardDemos

### The definitions

Two definitions, one per grid. The content grid has two
[overrides](/typed-grid-areas/concepts/overrides/): without a tag, and
without an icon but with a tag, where the derived layouts would stack what
should sit side by side.

```tsx
import type { ReactNode } from "react";
import { createGridComponent } from "@typed-grid-areas/react";
import { preferenceCardGrid, preferenceContentGrid } from "./designs";

const Content = createGridComponent(preferenceContentGrid);
const Card = createGridComponent(preferenceCardGrid, { nest: { content: Content } });

type PreferenceCardProps = {
  input?: ReactNode;
  icon?: ReactNode;
  title: ReactNode;
  tag?: ReactNode;
  children?: ReactNode;
};

export function PreferenceCard({ input, icon, title, tag, children }: PreferenceCardProps) {
  return (
    <Card className="card">
      {input && (
        <Card.input asChild className="panel">
          <label>{input}</label>
        </Card.input>
      )}
      <Card.content className="panel content">
        {icon && <Card.content.icon>{icon}</Card.content.icon>}
        <Card.content.information>{title}</Card.content.information>
        {tag && <Card.content.cardTag>{tag}</Card.content.cardTag>}
        {children && <Card.content.children>{children}</Card.content.children>}
      </Card.content>
    </Card>
  );
}
```

import { Component, input } from "@angular/core";
import { createGrid, TgaArea, TgaGrid, TgaSizeContainer } from "@typed-grid-areas/angular";
import { preferenceCardGrid, preferenceContentGrid } from "./designs";

const Content = createGrid(preferenceContentGrid);
const Card = createGrid(preferenceCardGrid, { nest: { content: Content } });

@Component({
  selector: "app-preference-card",
  imports: [TgaGrid, TgaArea, TgaSizeContainer],
  template: `
    <div [tgaGrid]="Card" class="card">
      @if (selectable()) {
        <label [tgaArea]="Card.input" class="panel"><input type="checkbox" /></label>
      }
      <!-- The content grid has breakpoints, so it sits in its own size container. -->
      <div tgaSizeContainer>
        <div [tgaGrid]="Card.content" class="panel content">
          @if (icon()) {
            <span [tgaArea]="Card.content.icon">{{ icon() }}</span>
          }
          <div [tgaArea]="Card.content.information">{{ heading() }}</div>
          @if (tag()) {
            <span [tgaArea]="Card.content.cardTag">{{ tag() }}</span>
          }
          @if (details()) {
            <p [tgaArea]="Card.content.children">{{ details() }}</p>
          }
        </div>
      </div>
    </div>
  `,
})
export class PreferenceCardComponent {
  protected readonly Card = Card;
  readonly selectable = input(false);
  readonly icon = input<string>();
  readonly heading = input.required<string>();
  readonly tag = input<string>();
  readonly details = input<string>();
}

<TabItem label="Lit">

```ts lit
import { html, LitElement, nothing } from "lit";
import { createGrid, gridStyles } from "@typed-grid-areas/lit";
import { preferenceCardGrid, preferenceContentGrid } from "./designs";

const Content = createGrid(preferenceContentGrid);
const Card = createGrid(preferenceCardGrid, { nest: { content: Content } });

class PreferenceCard extends LitElement {
  static styles = [gridStyles(Card)];
  static properties = {
    selectable: { type: Boolean },
    icon: {},
    heading: {},
    tag: {},
    details: {},
  };
  declare selectable: boolean;
  declare icon?: string;
  declare heading: string;
  declare tag?: string;
  declare details?: string;

  constructor() {
    super();
    this.selectable = false;
    this.heading = "";
  }

  render() {
    const { icon, tag, details } = this;
    return Card(
      html`
        ${this.selectable
          ? html`<label data-area=${Card.input} class="panel"><input type="checkbox" /></label>`
          : nothing}
        ${Card.content(
          html`
            ${icon ? html`<span data-area=${Card.content.icon}>${icon}</span>` : nothing}
            <div data-area=${Card.content.information}>${this.heading}</div>
            ${tag ? html`<span data-area=${Card.content.cardTag}>${tag}</span>` : nothing}
            ${details ? html`<p data-area=${Card.content.children}>${details}</p>` : nothing}
          `,
          { class: "panel content" },
        )}
      `,
      { class: "card" },
    );
  }
}

customElements.define("preference-card", PreferenceCard);
```

## ID card

The [layered](/typed-grid-areas/concepts/layers/) card. On narrow cards the
logo, the badge and the identity sit in the photo's corners, and the name
turns white over the photo; from `@md` (448px) the photo is a column of its
own and only the badge stays on it.

<IdCardDemos
  frames={[
    { label: "Everything", width: NARROW, present: ["logo", "qr", "badge"] },
    { label: "Everything", width: WIDE, present: ["logo", "qr", "badge"] },
    { label: "No logo, no QR code", width: NARROW, present: ["badge"] },
    { label: "No badge", width: WIDE, present: ["logo", "qr"] },
  ]}
/>

### The definition

<Code code={designSource("idCardGrid")} lang="ts" title="designs.ts" />

The name turns white over the photo with a container style query on
`--tga-over`, in your own CSS:

```css
@container style(--tga-over: photo) {
  .identity-name {
    color: white;
  }
}
```

import { createGridComponent } from "@typed-grid-areas/react";
import { idCardGrid } from "./designs";

const Card = createGridComponent(idCardGrid);

type Person = { name: string; role: string; photoUrl: string; verified?: boolean };

export function IdCard({ person, qrUrl }: { person: Person; qrUrl?: string }) {
  return (
    <Card className="id-card">
      <Card.photo asChild>
        <img src={person.photoUrl} alt="" />
      </Card.photo>
      <Card.logo>ACME</Card.logo>
      {person.verified && <Card.badge>✓</Card.badge>}
      <Card.identity>
        <span className="identity-name">{person.name}</span>
      </Card.identity>
      <Card.details>{person.role}</Card.details>
      {qrUrl && (
        <Card.qr asChild>
          <img src={qrUrl} alt="QR code" />
        </Card.qr>
      )}
    </Card>
  );
}
